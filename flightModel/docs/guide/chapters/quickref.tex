\chapter{Quick reference}\label{app:quickref}

\section{Commands}

\begin{shell}
source /usr/lib/openfoam/openfoam2412/etc/bashrc

# geometry and one mesh
python3 mesh/makeRocket.py mesh/stl/<name>.stl --finRoll <deg>
mesh/Allmesh ~/meshes/<mesh> [--maxCellSize <m>] [--nLayers <n>] [--thicknessRatio <r>]
./post/firstLayer.py ~/meshes/<mesh>

# one case
./newCase.sh ~/runs/<case> --mesh ~/meshes/<mesh> --solver pimple --Minf <M> --np <N>
cd ~/runs/<case> && ./Allrun

# study
./runMesh.sh meshes.txt ~/meshes
nohup ./runCase.sh <sweep-file> > ~/log.<sweep> 2>&1 &

# results
./post/monitor.sh ~/runs/<case>
./post/sweepTable.sh ~/runs > sweep.csv
python3 flightData/cdFlight.py

# binding of a running case
for p in $(pgrep -x rhoPimpleFoam); do taskset -cp $p; done
\end{shell}

\section{Run names}

\texttt{m<M>-a<alpha>-h<n>[-pimple]}: Mach number times ten, angle of attack in
degrees, mesh. \texttt{-pimple} marks the \cmd{rhoPimpleFoam} start run of a supersonic
chain.
